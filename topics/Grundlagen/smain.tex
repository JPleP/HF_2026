\Term{Matrixbeschreibungen}
\begin{pitemize}
    \item \Term{S-Parameter (Streuparameter)}:
    \begin{pitemize}
        \item Beschreibung über Wellen ($a_i$ (Eingang Oben), $b_i$ (Ausgang unter), Nummer sind die Seite)
        \item $\begin{pmatrix} b_1 \\ b_2 \end{pmatrix}
                =
                \begin{pmatrix} S_{11} & S_{12} \\ S_{21} & S_{22} \end{pmatrix}
                \begin{pmatrix} a_1 \\ a_2 \end{pmatrix}$
            \begin{pitemize}
                \item $S_{11} \big|_{a_2 = 0} = \frac{b_1}{a_1}$: Reflexion Eingang
                \item $S_{21} \big|_{a_2 = 0} = \frac{b_2}{a_1}$: Transmission (Verstärkung)
                \item $S_{12}$: Rückwirkung
                \item $S_{22}$: Reflexion Ausgang
            \end{pitemize}
        \item $\frac{\begin{pmatrix}
            A + B - C - D & 2\Delta_A \\
            2  & -A + B - C +D
        \end{pmatrix}}{A+B+C+D}$
        \item $\frac{\begin{pmatrix}
            \Sigma_{12} & \Delta_\Sigma \\
            1  & -\Sigma_{21}
        \end{pmatrix}}{\Sigma_{22}}$
        \item \bildf{Streuparameter}
    \end{pitemize}

    \item \Term{Impedanzparameter}:
    \begin{pitemize}
        \item $U_1 = Z_{11}I_1 + Z_{12}I_2$
        \item $U_2 = Z_{21}I_1 + Z_{22}I_2$
        \item \bildf{Impedanzparameter}
    \end{pitemize}

    \item \Term{Kettenparameter}:
    \begin{pitemize}
        \item Darstellung der ABCD-Parameter
        \item $\begin{pmatrix} U_1 \\ I_1 \end{pmatrix}
            =
            \begin{pmatrix} A & B \\ C & D \end{pmatrix}
            \begin{pmatrix} U_2 \\ -I_2 \end{pmatrix}$
        \item $\frac{\begin{pmatrix}
            -\Delta_S + S_{11} - S_{22} + 1 & \Delta_S + S_{11} + S_{22} + 1 \\
            \Delta_S - S_{11} - S_{22} + 1 & -\Delta_S - S_{11} + S_{22} + 1
        \end{pmatrix}}{2S_{21}}$
        \item \bildf{Kettenparameter}
    \end{pitemize}

    \item \Term{Transmissionsmatrix}:
    \begin{pitemize}
        \item Multiplikation Vereinfachung
        \item $\begin{pmatrix} b_1 \\ a_1 \end{pmatrix}
                =
                \begin{pmatrix} \Sigma_{11} & \Sigma_{12} \\ \Sigma_{21} & \Sigma_{22} \end{pmatrix}
                \begin{pmatrix} a_2 \\ b_2 \end{pmatrix}$
        \item $[\Sigma] = \frac{1\begin{pmatrix}
            -\Delta_S & S_{11} \\ -S_{22} & 1
        \end{pmatrix}}{S_{21}}$
        \item \bildf{Transmissionsparamter}
        \item \bildf{Hintereinanderschaltung}
        \item $\left[\Sigma_g\right] =  \left[\Sigma_1\right] \left[\Sigma_2\right]$   
    \end{pitemize}
    
\end{pitemize}
